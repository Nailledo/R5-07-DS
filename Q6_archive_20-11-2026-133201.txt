public class Q6
{
    public static void main(String[] args) 
	{
        int n = Integer.parseInt(args[0]);

        for (int i = 1; i <= n; i++) 
		{

            StringBuilder ligne = new StringBuilder();

            for (int j = i; j >= 1; j--) 
			{
                ligne.append("x");
                if (j > 1) ligne.append(" ");
            }
            System.out.println(ligne);
        }
    }
}
